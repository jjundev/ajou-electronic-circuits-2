=== 1. Loaded open-loop (R_F to ground at both N and P) ===
R_in,open  = R_F/(R_F*g_m1 + 1)    [= R_F || 1/g_m1 : True ]
R_out,open = R_F*R_M*r_O2/(R_F*R_M + R_F*r_O2 + R_M*r_O2)    [= r_O2 || R_M || R_F]
A_open     = -R_D*R_F**2*R_M*g_m1*g_m2*r_O2/((R_F*g_m1 + 1)*(R_F*R_M + R_F*r_O2 + R_M*r_O2))
A_open == -(R_F||1/gm1)*gm1*R_D*gm2*(rO2||R_M||R_F) : True
A_open == -[R_F*R_D/(R_F+1/gm1)]*gm2*(rO2||R_M||R_F) : True

=== 2. Feedback factor and loop gain ===
K = I_2/V_1 = -1/R_F
K*A_open    = R_D*R_F*R_M*g_m1*g_m2*r_O2/((R_F*g_m1 + 1)*(R_F*R_M + R_F*r_O2 + R_M*r_O2))
K*A_open > 0 (negative feedback): True

=== 3. Closed loop via 1 + K*A_open ===
1 + K*A_open = (R_D*R_F*R_M*g_m1*g_m2*r_O2 + R_F**2*R_M*g_m1 + R_F**2*g_m1*r_O2 + R_F*R_M*g_m1*r_O2 + R_F*R_M + R_F*r_O2 + R_M*r_O2)/((R_F*g_m1 + 1)*(R_F*R_M + R_F*r_O2 + R_M*r_O2))
A_closed     = -R_D*R_F**2*R_M*g_m1*g_m2*r_O2/(R_D*R_F*R_M*g_m1*g_m2*r_O2 + R_F**2*R_M*g_m1 + R_F**2*g_m1*r_O2 + R_F*R_M*g_m1*r_O2 + R_F*R_M + R_F*r_O2 + R_M*r_O2)
R_in,closed  = R_F*(R_F*R_M + R_F*r_O2 + R_M*r_O2)/(R_D*R_F*R_M*g_m1*g_m2*r_O2 + R_F**2*R_M*g_m1 + R_F**2*g_m1*r_O2 + R_F*R_M*g_m1*r_O2 + R_F*R_M + R_F*r_O2 + R_M*r_O2)
R_out,closed = R_F*R_M*r_O2*(R_F*g_m1 + 1)/(R_D*R_F*R_M*g_m1*g_m2*r_O2 + R_F**2*R_M*g_m1 + R_F**2*g_m1*r_O2 + R_F*R_M*g_m1*r_O2 + R_F*R_M + R_F*r_O2 + R_M*r_O2)

=== 4. Exact nodal KCL (R_F between N and P, finite r_O2) ===
A_exact = V_out/I_in = -R_M*r_O2*(R_D*R_F*g_m1*g_m2 - 1)/(R_D*R_M*g_m1*g_m2*r_O2 + R_F*R_M*g_m1 + R_F*g_m1*r_O2 + R_M*g_m1*r_O2 + R_M + r_O2)
R_in,exact = v_N/I_in = (R_F*R_M + R_F*r_O2 + R_M*r_O2)/(R_D*R_M*g_m1*g_m2*r_O2 + R_F*R_M*g_m1 + R_F*g_m1*r_O2 + R_M*g_m1*r_O2 + R_M + r_O2)
R_out,exact (I_in open) = R_M*r_O2*(R_F*g_m1 + 1)/(R_D*R_M*g_m1*g_m2*r_O2 + R_F*R_M*g_m1 + R_F*g_m1*r_O2 + R_M*g_m1*r_O2 + R_M + r_O2)
ratio A_exact/A_closed = (R_D*R_F*g_m1*g_m2 - 1)*(R_D*R_F*R_M*g_m1*g_m2*r_O2 + R_F**2*R_M*g_m1 + R_F**2*g_m1*r_O2 + R_F*R_M*g_m1*r_O2 + R_F*R_M + R_F*r_O2 + R_M*r_O2)/(R_D*R_F**2*g_m1*g_m2*(R_D*R_M*g_m1*g_m2*r_O2 + R_F*R_M*g_m1 + R_F*g_m1*r_O2 + R_M*g_m1*r_O2 + R_M + r_O2))
  = 1 - 1/(g_m1*g_m2*R_D*R_F) ? : False
